# Backend development environment.

# Install into a virtual environment from the repository root:
#     python -m venv .venv
#     .venv/Scripts/activate      (Windows)
#     source .venv/bin/activate   (macOS / Linux)
#     pip install -r requirements-dev.txt

# The editable installs must stay in dependency order: each package's
# dependencies have to be installed from a local path before the package that

# jip-ai-core; jip-api needs both; jip-worker needs jip-api.

-e ./packages/prompts
